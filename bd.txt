DROP DATABASE IF EXISTS coffee_shop;
CREATE DATABASE coffee_shop CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE coffee_shop;

-- =========================================
-- TABlES
-- =========================================

CREATE TABLE client(
    id_client INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    email VARCHAR(100)
);

CREATE TABLE category(
    id_category INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE drink(
    id_drink INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    stock INT NOT NULL,
    id_category INT NOT NULL,
    FOREIGN KEY(id_category) REFERENCES category(id_category)
);

CREATE TABLE sell(
    id_sell INT AUTO_INCREMENT PRIMARY KEY,
    date_sell DATETIME DEFAULT CURRENT_TIMESTAMP,
    id_client INT NOT NULL,
    total DECIMAL(10,2) DEFAULT 0,
    FOREIGN KEY(id_client) REFERENCES client(id_client)
);

CREATE TABLE item_sell(
    id_item INT AUTO_INCREMENT PRIMARY KEY,
    id_sell INT NOT NULL,
    id_drink INT NOT NULL,
    quantity INT NOT NULL,
    unitary_value DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    FOREIGN KEY(id_sell) REFERENCES sell(id_sell),
    FOREIGN KEY(id_drink) REFERENCES drink(id_drink)
);

-- =========================================
-- TRIGGER
-- Only positive values
-- =========================================

DELIMITER //

CREATE TRIGGER trg_drink_before_update
BEFORE UPDATE ON drink
FOR EACH ROW
BEGIN
    IF NEW.price<0 THEN
        SET NEW.preco=ABS(NEW.preco);
    END IF;

    IF NEW.estoque<0 THEN
        SET NEW.estoque=ABS(NEW.estoque);
    END IF;
END//

DELIMITER ;

-- =========================================
-- FUNÇÃO
-- Calcula subtotal
-- =========================================

DELIMITER //

CREATE FUNCTION fn_calcular_subtotal(qtd INT,valor DECIMAL(10,2))
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    RETURN qtd*valor;
END//

DELIMITER ;

-- =========================================
-- STORED PROCEDURE
-- Buscar bebidas
-- =========================================

DELIMITER //

CREATE PROCEDURE sp_buscar_bebidas(IN nomeBusca VARCHAR(100))
BEGIN
    SELECT
        b.id_bebida,
        b.nome,
        b.preco,
        b.estoque,
        c.nome AS categoria
    FROM bebida b
    JOIN categoria c
        ON b.id_categoria=c.id_categoria
    WHERE b.nome LIKE CONCAT('%',nomeBusca,'%')
    ORDER BY b.nome;
END//

DELIMITER ;

-- =========================================
-- PROCEDURE DASHBOARD
-- =========================================

DELIMITER //

CREATE PROCEDURE sp_dashboard()
BEGIN
    SELECT
        (SELECT COUNT(*) FROM venda) AS total_vendas,
        (SELECT COUNT(*) FROM cliente) AS total_clientes,
        (SELECT ROUND(SUM(total),2) FROM venda) AS faturamento_total,
        (SELECT ROUND(AVG(total),2) FROM venda) AS ticket_medio;
END//

DELIMITER ;

-- =========================================
-- VIEW COMPLETA
-- =========================================

CREATE VIEW vw_vendas_completas AS
SELECT
    v.id_venda,
    v.data_venda,
    cl.nome AS cliente,
    b.nome AS bebida,
    cat.nome AS categoria,
    i.quantidade,
    i.valor_unitario,
    i.subtotal
FROM venda v
JOIN cliente cl ON v.id_cliente=cl.id_cliente
JOIN item_venda i ON v.id_venda=i.id_venda
JOIN bebida b ON i.id_bebida=b.id_bebida
JOIN categoria cat ON b.id_categoria=cat.id_categoria;

-- =========================================
-- VIEW ESTOQUE BAIXO
-- =========================================

CREATE VIEW vw_estoque_baixo AS
SELECT
    nome,
    estoque,
    preco
FROM bebida
WHERE estoque<50;

-- =========================================
-- CONSULTA CTE
-- Produtos mais vendidos
-- =========================================

WITH ranking AS(
    SELECT
        b.nome,
        SUM(i.quantidade) AS quantidade_vendida
    FROM item_venda i
    JOIN bebida b
        ON i.id_bebida=b.id_bebida
    GROUP BY b.nome
)
SELECT *
FROM ranking
ORDER BY quantidade_vendida DESC;
